# Docker Compose başlamak için:
# docker-compose up --build

# Backend'i tekrar başlatmak için:
# docker-compose restart backend

# Logları görmek için:
# docker-compose logs -f backend

# Veritabanına bağlanmak için:
# docker-compose exec postgres psql -U michat -d michat_db

# Tüm servisleri durdurmak için:
# docker-compose down

# Tüm verileri silmek için:
# docker-compose down -v
